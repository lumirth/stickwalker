cmake_minimum_required(VERSION 3.16)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" release LIMIT_COUNT 1)
if(DEFINED CACHE{STICKWALKER_VERSION})
    set(PROJECT_VER "${STICKWALKER_VERSION}")
    unset(STICKWALKER_VERSION CACHE)
else()
    set(PROJECT_VER "${release}-dev")
endif()
string(LENGTH "${PROJECT_VER}" length)
string(FIND "${PROJECT_VER}" "${release}-dev" dev)
if(length GREATER 31 OR NOT (PROJECT_VER STREQUAL release OR dev EQUAL 0))
    message(FATAL_ERROR "Firmware version ${PROJECT_VER} must be ${release} or ${release}-dev, within 31 bytes.")
endif()
set(EXTRA_COMPONENT_DIRS "pokewalker")
set(COMPONENTS main)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(stickwalker)
